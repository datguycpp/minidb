-- Correlated scalar subquery in the projection: per-row lookup against the
-- other table — the classic form, outer column referenced inside.

CREATE TABLE a (id INTEGER, name TEXT);
CREATE TABLE b (aid INTEGER, score INTEGER);
INSERT INTO a VALUES (1, 'x');
INSERT INTO a VALUES (2, 'y');
INSERT INTO b VALUES (1, 10);
INSERT INTO b VALUES (1, 20);
SELECT name, (SELECT MAX(score) FROM b WHERE b.aid = a.id) AS top FROM a ORDER BY name;

-- Correlated subquery in WHERE: rows whose own-group maximum they equal.

CREATE TABLE a (id INTEGER, v INTEGER);
INSERT INTO a VALUES (1, 5);
INSERT INTO a VALUES (1, 9);
INSERT INTO a VALUES (2, 3);
SELECT * FROM a WHERE v = (SELECT MAX(v) FROM a AS x WHERE x.id = a.id) ORDER BY id, v;

-- Correlated EXISTS: rows in a that have any match in b.

CREATE TABLE a (id INTEGER);

SELECT * FROM a WHERE EXISTS (SELECT 1 FROM b WHERE b.aid = a.id) ORDER BY id;
