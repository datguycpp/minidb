-- FTS: create virtual table, insert, MATCH a plain term.

CREATE VIRTUAL TABLE docs USING fts5(body);
INSERT INTO docs VALUES ('the quick brown fox');
INSERT INTO docs VALUES ('a lazy dog sleeps');
SELECT body FROM docs WHERE docs MATCH 'fox';

-- MATCH with AND / OR.

CREATE VIRTUAL TABLE docs USING fts5(body);
INSERT INTO docs VALUES ('quick fox');
INSERT INTO docs VALUES ('lazy dog');
INSERT INTO docs VALUES ('quick dog');
SELECT body FROM docs WHERE docs MATCH 'quick AND dog' ORDER BY body;

CREATE VIRTUAL TABLE docs USING fts5(body);
INSERT INTO docs VALUES ('quick fox');
INSERT INTO docs VALUES ('lazy dog');
SELECT body FROM docs WHERE docs MATCH 'quick OR lazy' ORDER BY body;

-- MATCH with NOT.

CREATE VIRTUAL TABLE docs USING fts5(body);
INSERT INTO docs VALUES ('quick fox');
INSERT INTO docs VALUES ('quick dog');
SELECT body FROM docs WHERE docs MATCH 'quick NOT fox';

-- Column filter: term restricted to one column.

CREATE VIRTUAL TABLE docs USING fts5(title, body);
INSERT INTO docs VALUES ('fox hunting', 'an article about dogs');
INSERT INTO docs VALUES ('dog training', 'an article about foxes');
SELECT body FROM docs WHERE docs MATCH 'title:dog';

-- Trailing-prefix match: 'qui*' matches 'quick'.

CREATE VIRTUAL TABLE docs USING fts5(body);
INSERT INTO docs VALUES ('the quick brown fox');
INSERT INTO docs VALUES ('nothing here');
SELECT body FROM docs WHERE docs MATCH 'qui*';
