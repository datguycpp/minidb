-- UNION deduplicates across inputs; column names come from the FIRST select.

CREATE TABLE t1 (v INTEGER);
CREATE TABLE t2 (v INTEGER);
INSERT INTO t1 VALUES (1);
INSERT INTO t1 VALUES (2);
INSERT INTO t2 VALUES (2);
INSERT INTO t2 VALUES (3);
SELECT v AS val FROM t1 UNION SELECT v FROM t2 ORDER BY val;

-- UNION ALL keeps multiplicity: input duplicates survive.

CREATE TABLE t1 (v INTEGER);
INSERT INTO t1 VALUES (1);
INSERT INTO t1 VALUES (1);
INSERT INTO t1 VALUES (2);
SELECT v FROM t1 UNION ALL SELECT v FROM t1 ORDER BY v;

-- INTERSECT keeps only common rows.

INSERT INTO t1 VALUES (2);
INSERT INTO t2 VALUES (2);
INSERT INTO t2 VALUES (3);
SELECT v FROM t1 INTERSECT SELECT v FROM t2 ORDER BY v;

-- RULE (engineer's ask 3): branch column-count mismatch. Exact error text

-- PRECEDENCE — corrected after the M13 landing: SQLite does NOT bind
-- INTERSECT tighter (the SQL-standard reading this comment originally
-- stated). The reference evaluates compounds FLAT LEFT-ASSOCIATIVE:
-- ((t1 UNION t2) INTERSECT t3). With t1={1,2}, t2={2,3}, t3={2} that
-- gives {2}; the standard's t1 UNION (t2 INTERSECT t3) would give {1,2}.
-- The binary settled it (M13 report, case rows [(2,)]). Comment-only
-- correction; expected score delta 0 — expected values are computed at
-- run time, so this prose misled the builder, not the score.

CREATE TABLE t1 (v INTEGER);
CREATE TABLE t2 (v INTEGER);
CREATE TABLE t3 (v INTEGER);
INSERT INTO t1 VALUES (1);
INSERT INTO t1 VALUES (2);
INSERT INTO t2 VALUES (2);
INSERT INTO t2 VALUES (3);
INSERT INTO t3 VALUES (2);
SELECT v FROM t1 UNION SELECT v FROM t2 INTERSECT SELECT v FROM t3 ORDER BY v;

-- EXCEPT is left-associative: (t1 \ t2) \ t3 = {1};
-- right-assoc t1 \ (t2 \ t3) = {1,2,3}.

CREATE TABLE t1 (v INTEGER);
CREATE TABLE t2 (v INTEGER);
CREATE TABLE t3 (v INTEGER);
INSERT INTO t1 VALUES (1);
INSERT INTO t1 VALUES (2);
INSERT INTO t1 VALUES (3);
INSERT INTO t2 VALUES (2);
INSERT INTO t3 VALUES (3);
SELECT v FROM t1 EXCEPT SELECT v FROM t2 EXCEPT SELECT v FROM t3 ORDER BY v;

-- ENGINEER'S ASK 1+2: UNION ALL under a mixed-direction multi-key ORDER BY
-- over the whole compound.

CREATE TABLE t1 (a INTEGER, b INTEGER);
CREATE TABLE t2 (a INTEGER, b INTEGER);
INSERT INTO t1 VALUES (1, 2);
INSERT INTO t1 VALUES (1, 1);
INSERT INTO t2 VALUES (1, 3);
INSERT INTO t2 VALUES (2, 0);
SELECT a, b FROM t1 UNION ALL SELECT a, b FROM t2 ORDER BY a ASC, b DESC;

-- RULE (engineer's ask 3, the ORDER-BY variant): ORDER BY naming a column
-- outside the compound's result set. Exact refusal text = reference's.

CREATE TABLE t1 (a INTEGER, b INTEGER);
CREATE TABLE t2 (a INTEGER, b INTEGER);
INSERT INTO t1 VALUES (1, 10);
INSERT INTO t2 VALUES (1, 20);
SELECT a FROM t1 UNION SELECT a FROM t2 ORDER BY b;

-- UNION treats NULLs as equal for deduplication: two NULL rows collapse.

CREATE TABLE t (v INTEGER);
INSERT INTO t VALUES (NULL);
INSERT INTO t VALUES (NULL);
SELECT v FROM t UNION SELECT v FROM t ORDER BY v;

-- INTERSECT with NULL on both sides: common, so the row is kept.

CREATE TABLE t1 (v INTEGER);
CREATE TABLE t2 (v INTEGER);
INSERT INTO t1 VALUES (NULL);
INSERT INTO t2 VALUES (NULL);
SELECT v FROM t1 INTERSECT SELECT v FROM t2;

-- EXCEPT removes a NULL-matching row: NULL equals NULL in set operations.

CREATE TABLE t1 (v INTEGER);
CREATE TABLE t2 (v INTEGER);
INSERT INTO t1 VALUES (1);
INSERT INTO t1 VALUES (NULL);
INSERT INTO t2 VALUES (NULL);
SELECT v FROM t1 EXCEPT SELECT v FROM t2 ORDER BY v;

-- Mixed storage classes through UNION; ORDER BY 1 orders by SQLite's
-- storage-class rule (numbers before text).

CREATE TABLE nums (n INTEGER);
CREATE TABLE words (s TEXT);
INSERT INTO nums VALUES (10);
INSERT INTO nums VALUES (2);
INSERT INTO words VALUES ('a');
INSERT INTO words VALUES ('b');
SELECT n FROM nums UNION SELECT s FROM words ORDER BY 1;

-- INTERSECT ALL: SQLite's documented compound set does not include it —
-- if the reference rejects, we reject identically; if the reference
-- accepts, conformance means accepting. Runtime decides, either way owned.

CREATE TABLE t1 (v INTEGER);
CREATE TABLE t2 (v INTEGER);
INSERT INTO t1 VALUES (1);
INSERT INTO t1 VALUES (1);
INSERT INTO t2 VALUES (1);
SELECT v FROM t1 INTERSECT ALL SELECT v FROM t2;

-- Compound inside a derived table (M8-era machinery) with an outer filter.

CREATE TABLE t1 (v INTEGER);
CREATE TABLE t2 (v INTEGER);
INSERT INTO t1 VALUES (1);
INSERT INTO t1 VALUES (3);
INSERT INTO t2 VALUES (2);
INSERT INTO t2 VALUES (3);
SELECT * FROM (SELECT v FROM t1 UNION SELECT v FROM t2) AS u WHERE u.v > 1 ORDER BY u.v;

-- Compound as an IN subquery.

CREATE TABLE t1 (v INTEGER);
CREATE TABLE t2 (v INTEGER);
CREATE TABLE probe (v INTEGER);
INSERT INTO t1 VALUES (1);
INSERT INTO t2 VALUES (2);
INSERT INTO probe VALUES (1);
INSERT INTO probe VALUES (4);
SELECT * FROM probe WHERE v IN (SELECT v FROM t1 UNION SELECT v FROM t2) ORDER BY v;

-- Empty right side: UNION with an empty input yields the left side.

CREATE TABLE t1 (v INTEGER);
CREATE TABLE t2 (v INTEGER);
INSERT INTO t1 VALUES (1);
INSERT INTO t1 VALUES (2);
SELECT v FROM t1 UNION SELECT v FROM t2 ORDER BY v;

-- Compound with WHERE on both branches, UNION ALL ordering by alias of
-- the first branch.

CREATE TABLE t1 (v INTEGER);
CREATE TABLE t2 (v INTEGER);
INSERT INTO t1 VALUES (3);
INSERT INTO t1 VALUES (1);
INSERT INTO t2 VALUES (2);
SELECT v AS k FROM t1 WHERE v > 1 UNION ALL SELECT v FROM t2 ORDER BY k;
