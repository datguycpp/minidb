-- [pin] -> yields TEXT, ->> yields the SQL value (typeof proves it).

SELECT typeof('{"a":5}' -> '$.a'), typeof('{"a":5}' ->> '$.a');

-- [pin] Precedence is not C-like: || binds tighter than -> ->>, which
-- binds tighter than * /. ('a'||1) is text 'a1', times 2 truncates to 0.

SELECT 'a' || 1 * 2;

-- [pin] Arrow binds tighter than *: 2 * (value 3) = 6.

SELECT 2 * '{"a":3}' ->> '$.a';

-- [pin] JSON null becomes SQL NULL.

SELECT json_extract('{"a":null}', '$.a') IS NULL;
