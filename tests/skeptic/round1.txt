-- S1 nested subqueries: depth-4 correlation, innermost references outermost.
CREATE TABLE a (id INTEGER, k INTEGER);
CREATE TABLE b (k INTEGER);
CREATE TABLE c (k INTEGER);
CREATE TABLE d (k INTEGER);
INSERT INTO a VALUES (1, 10), (2, 20);
INSERT INTO b VALUES (10), (20);
INSERT INTO c VALUES (10), (20);
INSERT INTO d VALUES (10), (20);
SELECT a.id, (SELECT COUNT(*) FROM b WHERE b.k = a.k AND EXISTS (SELECT 1 FROM c WHERE c.k = b.k AND EXISTS (SELECT 1 FROM d WHERE d.k = c.k AND d.k = a.k))) FROM a ORDER BY a.id;

-- S2 correlated scalar subquery inside ORDER BY (values unique: well-defined).
CREATE TABLE t (id INTEGER, k INTEGER);
CREATE TABLE u (k INTEGER, v INTEGER);
INSERT INTO t VALUES (1, 20), (2, 10), (3, 30);
INSERT INTO u VALUES (10, 100), (20, 200), (30, 300);
SELECT t.id FROM t ORDER BY (SELECT u.v FROM u WHERE u.k = t.k) DESC, t.id;

-- S3 correlated EXISTS containing a JOIN.
CREATE TABLE t (k INTEGER);
CREATE TABLE u (k INTEGER, w INTEGER);
CREATE TABLE v (k INTEGER);
INSERT INTO t VALUES (1), (2);
INSERT INTO u VALUES (1, 5), (2, 6);
INSERT INTO v VALUES (1), (3);
SELECT t.k FROM t WHERE EXISTS (SELECT 1 FROM u JOIN v ON u.k = v.k WHERE u.k = t.k) ORDER BY t.k;

-- S4 correlated subquery inside HAVING.
CREATE TABLE t (k INTEGER, g INTEGER);
CREATE TABLE m (g INTEGER, cap INTEGER);
INSERT INTO t VALUES (1, 1), (2, 1), (3, 2), (4, 2), (5, 2);
INSERT INTO m VALUES (1, 1), (2, 2);
SELECT g, COUNT(*) FROM t GROUP BY g HAVING COUNT(*) <= (SELECT cap FROM m WHERE m.g = t.g) ORDER BY g;

-- S5 correlated FROM-subquery: no LATERAL in SQL, both should refuse.
CREATE TABLE t (k INTEGER);
INSERT INTO t VALUES (1), (2);
SELECT * FROM t, (SELECT k + t.k) AS d;

-- S6 scalar subquery with ORDER BY LIMIT (single row, well-defined).
CREATE TABLE u (k INTEGER);
INSERT INTO u VALUES (3), (1), (2);
SELECT (SELECT k FROM u ORDER BY k LIMIT 1) AS first, (SELECT MAX(k) FROM u) AS mx;

-- S7 LEFT JOIN with NULL keys: NULL never matches, unmatched rows pad.
CREATE TABLE t (k INTEGER, a TEXT);
CREATE TABLE u (k INTEGER, b TEXT);
INSERT INTO t VALUES (1, 'x'), (2, 'y'), (NULL, 'z');
INSERT INTO u VALUES (1, 'p'), (NULL, 'q');
SELECT t.a, u.b FROM t LEFT JOIN u ON t.k = u.k ORDER BY t.a;

-- S8 NOT IN over a subquery containing NULL: three-valued logic empties it.
CREATE TABLE t (k INTEGER);
CREATE TABLE u (k INTEGER);
INSERT INTO t VALUES (1), (2), (3);
INSERT INTO u VALUES (1), (NULL);
SELECT k FROM t WHERE k NOT IN (SELECT k FROM u) ORDER BY k;

-- S9 JOIN ON with NULL = NULL never matches (visible through LEFT JOIN).

-- S10 COUNT(*) vs COUNT(column) over LEFT JOIN: NULL pads do not count.

-- S18 NOCASE equality inside a join condition.

-- S19 blob = text is false in a join; observable through typeof.
CREATE TABLE t (v);
CREATE TABLE u (v);
INSERT INTO t VALUES (X'41'), ('A');
INSERT INTO u VALUES ('A');
SELECT typeof(t.v) AS ty FROM t JOIN u ON t.v = u.v ORDER BY ty;
