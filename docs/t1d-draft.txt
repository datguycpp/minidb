-- Rollback discards an insert.

CREATE TABLE t (id INTEGER);
BEGIN;
INSERT INTO t VALUES (1);
ROLLBACK;
SELECT * FROM t;

-- Commit keeps an insert.

CREATE TABLE t (id INTEGER);
BEGIN;
INSERT INTO t VALUES (1);
COMMIT;
SELECT * FROM t;

-- Rollback discards DDL: the dropped table must be missing on BOTH sides,
-- with byte-identical error text (error-agreement case).

-- Rollback leaves rows written before the transaction.
